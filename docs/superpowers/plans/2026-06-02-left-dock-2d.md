# Left Dock (2D) + Tool-Activation Convergence — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Converge every 2D tool-activation site on one writer (`activate2DTool`) so tools actually activate, then replace the broken `FloatingToolbar` with the redesign's left (2D) dock built on that writer — surfacing a `Sew` button that closes the 2D-sew discoverability gap.

**Architecture:** A plain helper `activate2DTool(dispatch, id)` dispatches both `setActiveTool` (editorSlice — the drawing-engine truth) and `setActiveTool2D` (toolSlice — the per-tool-options key + the `measure`/`walkSeam` gate). All 2D activation sites (keyboard, the new dock, the empty-state Draw button, the agent's `activateTool`) route through it — fixing two latent single-slice bugs. The dock is a presentational single-select `Dock` + a `dock2dTools` registry + a `LeftDock` container.

**Tech Stack:** React + TypeScript, Redux Toolkit (`editorSlice`/`toolSlice`), Vitest + @testing-library/react, lucide-react icons, CSS with the global dark-chrome tokens, Playwright (live verification).

**Spec:** `docs/superpowers/specs/2026-06-02-left-dock-2d-design.md`

```ts
import type { Dispatch } from '@reduxjs/toolkit';
import { setActiveTool } from './editorSlice';
import { setActiveTool2D } from './toolSlice';
import type { Tool2D } from '../types/types';

/**
 * The single writer for 2D tool activation.
 *
 * `editor.activeTool` is the drawing-engine truth (what a canvas click does);
 * `tools.activeTool2D` is the per-tool-options key + the gate for the two tools
 * (measure, walkSeam) built to read it. They are different facts, not copies —
 * so every activation site MUST set both, or a tool silently half-activates.
 * Route all 2D tool activation through here instead of hand-writing the pair.
 *
 * `id` is `string` (not `Tool2D`) because legacy bindings activate ids that
 * aren't in the union (circle/trace/dart/pin); the cast is centralized here.
 */
export function activate2DTool(dispatch: Dispatch, id: string): void {
  dispatch(setActiveTool(id as Tool2D));
  dispatch(setActiveTool2D(id));
}
```

```tsx
import type { DockToolDef } from './dock2dTools';
import './Dock.css';

interface DockProps {
  /** Domain anchor, e.g. "2D". Label only, not a button. */
  domainLabel: string;
  tools: DockToolDef[];
  /** The currently active tool id (the engine truth). */
  activeToolId: string;
  onSelectTool: (id: string) => void;
}

/**
 * A vertical single-select tool column (the redesign's side dock, left/2D use).
 * Presentational only — no store. The container supplies tools + active id and
 * handles activation. (Not generalized for the 3D side, whose affordances are
 * toggles + sim transport, not single-select — see the spec.)
 */
export default function Dock({ domainLabel, tools, activeToolId, onSelectTool }: DockProps) {
  return (
    <div className="dock">
      <div className="dock__label">{domainLabel}</div>
      {tools.map(tool => {
        const Icon = tool.icon;
        const active = tool.id === activeToolId;
        return (
          <button
            key={tool.id}
            type="button"
            className={`dock__tool ${active ? 'dock__tool--active' : ''}`}
            onClick={() => onSelectTool(tool.id)}
            title={`${tool.label} · ${tool.shortcut}`}
            aria-label={`${tool.label} · ${tool.shortcut}`}
            aria-pressed={active}
          >
            <Icon size={16} />
          </button>
        );
      })}
    </div>
  );
}
```

```tsx
import { useSelector, useDispatch } from 'react-redux';
import type { RootState } from '../../store';
import { activate2DTool } from '../../store/toolActivation';
import Dock from './Dock';
import { DOCK_2D_TOOLS } from './dock2dTools';

/** The left (2D) dock. Hidden in 3D-only view. */
export default function LeftDock() {
  const dispatch = useDispatch();
  const activeTool = useSelector((s: RootState) => s.editor.activeTool);
  const viewMode = useSelector((s: RootState) => s.editor.viewMode);

  if (viewMode === '3d') return null;

  return (
    <Dock
      domainLabel="2D"
      tools={DOCK_2D_TOOLS}
      activeToolId={activeTool}
      onSelectTool={(id) => activate2DTool(dispatch, id)}
    />
  );
}
```
