# Homepage de Vina — plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Publicar la home de `docs/superpowers/specs/2026-09-28-homepage-design.md` en `/` y mover el taller a `/app/`, con el patrón real de María en el hero, cero JavaScript y ≤ 500 KB.

**Architecture:** Segunda entrada HTML del mismo `vite build` (`index.html` = home, `app/index.html` = taller). La home es HTML + CSS estático en `home/`, con sus propios tokens. El SVG del hero lo genera un script determinista desde el DXF de María y un plugin de Vite de pocas líneas lo incrusta en el HTML (así el SVG usa los tokens y la fuente de la página). Los activos raster (figuras, póster, favicons, OG) los genera un solo script de Playwright contra el build de producción.

**Tech Stack:** Vite 6 (MPA), HTML/CSS sin framework, Playwright 1.58 (+ `@axe-core/playwright`), vitest, `tsx` para los scripts que importan el parser TS.

**Spec:** `docs/superpowers/specs/2026-09-28-homepage-design.md` (con la enmienda del 2026-09-29 al final, que manda sobre el cuerpo donde choquen).

```ts
// server/rutas.ts
import path from 'node:path';

export type Ruta =
  | { tipo: 'archivo'; ruta: string }
  | { tipo: 'redirigir'; a: string }
  | { tipo: 'no-encontrado' };

/** Qué servir de dist/ para un pathname. `/` es la home; bajo `/app/` el
 *  fallback SPA es el taller; fuera de ahí, lo que no existe es 404. */
export function resolverRuta(pathname: string, existe: (rel: string) => boolean): Ruta {
  const rel = path.posix.normalize(decodeURIComponent(pathname)).replace(/^\/+/, '');
  if (rel.startsWith('..')) return { tipo: 'no-encontrado' };
  if (rel === '' ) return { tipo: 'archivo', ruta: 'index.html' };
  if (rel === 'app') return { tipo: 'redirigir', a: '/app/' };
  if (rel.startsWith('app/')) {
    return rel !== 'app/' && existe(rel) ? { tipo: 'archivo', ruta: rel } : { tipo: 'archivo', ruta: 'app/index.html' };
  }
  return existe(rel) ? { tipo: 'archivo', ruta: rel } : { tipo: 'no-encontrado' };
}
```

```ts
// tests/repo/home-tokens.test.ts
import { describe, it, expect } from 'vitest';
import { readFileSync, existsSync } from 'node:fs';

const tokensCss = readFileSync('home/tokens.css', 'utf8');
const homeCss = existsSync('home/home.css') ? readFileSync('home/home.css', 'utf8') : '';
const html = existsSync('index.html') ? readFileSync('index.html', 'utf8') : '';

const tokens = Object.fromEntries(
  [...tokensCss.matchAll(/(--[\w-]+)\s*:\s*([^;]+);/g)].map((m) => [m[1], m[2].trim()]),
);
function lum(hex: string) {
  const n = hex.replace('#', '');
  const [r, g, b] = [0, 2, 4].map((i) => parseInt(n.slice(i, i + 2), 16) / 255)
    .map((c) => (c <= 0.03928 ? c / 12.92 : ((c + 0.055) / 1.055) ** 2.4));
  return 0.2126 * r + 0.7152 * g + 0.0722 * b;
}
const contraste = (a: string, b: string) => {
  const [x, y] = [lum(tokens[a]), lum(tokens[b])].sort((p, q) => q - p);
  return (x + 0.05) / (y + 0.05);
};

describe('home/tokens.css', () => {
  it.each([
    ['--tinta', '--papel', 14.5],
    ['--tinta-70', '--papel', 4.5],
    ['--senal-texto', '--papel', 4.5],
    ['--blanco', '--senal', 4.5],
    ['--papel', '--tinta', 14.5],
    ['--papel-60', '--tinta', 4.5],
    ['--tinta-70', '--papel-hundido', 4.5],
  ])('%s sobre %s ≥ %s:1', (texto, fondo, umbral) => {
    expect(contraste(texto, fondo)).toBeGreaterThanOrEqual(umbral);
  });

  it('el oro y el gris de verdad no alcanzan como texto (por eso están prohibidos)', () => {
    expect(contraste('--oro', '--papel')).toBeLessThan(3);
    expect(contraste('--gris', '--papel')).toBeLessThan(3);
  });

  it('solo Inter y JetBrains Mono', () => {
    expect(tokens['--font-sans']).toMatch(/^'Inter Variable'/);
    expect(tokens['--font-mono']).toMatch(/^'JetBrains Mono Variable'/);
    const familias = [...(homeCss + html).matchAll(/font-family\s*:\s*([^;]+);/g)].map((m) => m[1]);
    for (const f of familias) expect(f).toMatch(/^(var\(--font-(sans|mono)\)|'Inter Variable'|'JetBrains Mono Variable')$/);
  });
});

describe('home/home.css respeta los tokens', () => {
  it('sin var(--x, fallback)', () => expect(homeCss).not.toMatch(/var\(--[\w-]+\s*,/));
  it('sin colores literales', () => expect(homeCss).not.toMatch(/#[0-9a-fA-F]{3,8}\b|rgba?\(|hsla?\(/));
  it('el oro y el gris nunca como color de texto', () => {
    expect(homeCss).not.toMatch(/(^|[;{\s])color\s*:\s*var\(--(oro|gris)\)/m);
  });
  it('sin sombras ni clamp()', () => {
    expect(homeCss).not.toMatch(/box-shadow\s*:\s*(?!none)/);
    expect(homeCss).not.toMatch(/clamp\(/);
  });
});
```

```js
#!/usr/bin/env node
/**
 * DXF de María → home/img/patron.svg (el patrón del hero de la home).
 * Determinista: mismo DXF, mismo SVG byte a byte (tests/repo/home-patron.test.ts
 * lo compara con el commiteado). Spec: docs/superpowers/specs/2026-09-28-homepage-design.md
 * § 5.1 y su enmienda del 2026-09-29: sin línea de costura (el contorno ES la de
 * corte), sin pinza (el DXF no tiene), rótulos solo con lo que los datos sostienen.
 *
 *   npx tsx scripts/home/dxf-a-svg.mjs
 */
import { readFileSync, writeFileSync } from 'node:fs';
import { fileURLToPath } from 'node:url';
import { importDXFContent } from '../../src/utils/dxf/importPipeline.ts';

const W = 720, H = 540, MARGEN = 24, HUECO = 32, ALTO_COTA = 44, ALTO_ROTULO = 28;
const f = (n) => (Math.round(n * 10) / 10).toString();

const puntos = (outline) => outline.flatMap((c) => {
  const p = []; for (let i = 0; i < c.values.length; i += 2) p.push([c.values[i], c.values[i + 1]]); return p;
});
const caja = (pts) => {
  const xs = pts.map((p) => p[0]), ys = pts.map((p) => p[1]);
  return { x0: Math.min(...xs), y0: Math.min(...ys), x1: Math.max(...xs), y1: Math.max(...ys) };
};

/** Profundidad del escote: cuánto baja el borde de arriba en el tercio central.
 *  El delantero es el de escote más hondo. En coordenadas del importador
 *  (y hacia abajo en pantalla = y mayor). Verificar a ojo en Step 4. */
function escote(pieza) {
  const pts = puntos(pieza.outline), c = caja(pts);
  const ancho = c.x1 - c.x0, cx = (c.x0 + c.x1) / 2;
  const centro = pts.filter(([x]) => Math.abs(x - cx) < ancho / 6);
  return Math.min(...centro.map(([, y]) => y)) - c.y0;
}

export function generarPatronSvg(dxfTexto) {
  const { pieces, format } = importDXFContent(dxfTexto);
  const grandes = pieces.filter((p) => { const c = caja(puntos(p.outline)); return c.y1 - c.y0 > 500; });
  const delantero = [...grandes].sort((a, b) => escote(b) - escote(a))[0];
  const mangas = pieces.filter((p) => { const c = caja(puntos(p.outline)); return Math.abs((c.x1 - c.x0) - 381) < 20; });
  const manga = [...mangas].sort((a, b) => a.name.localeCompare(b.name))[0];

  const cd = caja(puntos(delantero.outline)), cm = caja(puntos(manga.outline));
  const s = Math.min(
    (W - 2 * MARGEN - HUECO) / ((cd.x1 - cd.x0) + (cm.x1 - cm.x0)),
    (H - ALTO_ROTULO - ALTO_COTA - MARGEN) / (cd.y1 - cd.y0),
  );
  const oy = ALTO_ROTULO + 8;
  const tD = ([x, y]) => [MARGEN + (x - cd.x0) * s, oy + (y - cd.y0) * s];
  const mx = MARGEN + (cd.x1 - cd.x0) * s + HUECO;
  const tM = ([x, y]) => [mx + (x - cm.x0) * s, oy + (y - cm.y0) * s];

  const d = (outline, t) => outline.map((c) => {
    if (c.type === 'Z') return 'Z';
    const [x, y] = t([c.values[0], c.values[1]]); return `${c.type}${f(x)} ${f(y)}`;
  }).join(' ');
  // internals[].path son comandos {type:'M'|'L', values:[x, y]} (medido 2026-09-29).
  const lineas = (pieza, t, tipo) => pieza.internals.filter((l) => l.type === tipo)
    .map((l) => { const [a, b] = [t(l.path[0].values), t(l.path[1].values)];
      return `<path class="${tipo}" d="M${f(a[0])} ${f(a[1])} L${f(b[0])} ${f(b[1])}"/>`; }).join('');

  // Dobladillo: el borde de abajo del delantero (puntos a ≤ 1 mm del y máximo).
  const bajos = puntos(delantero.outline).filter(([, y]) => cd.y1 - y <= 1);
  const dobladilloMm = Math.round(Math.max(...bajos.map((p) => p[0])) - Math.min(...bajos.map((p) => p[0])));
  const [hx0] = tD([Math.min(...bajos.map((p) => p[0])), cd.y1]);
  const [hx1, hy] = tD([Math.max(...bajos.map((p) => p[0])), cd.y1]);
  const yc = hy + 22;

  const archivo = `${delantero.name.replace(/-\d+$/, '')} · ${(format ?? 'dxf').toUpperCase()} · mm`;
  const svg = [
    `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${W} ${H}" class="patron" role="img" aria-label="El delantero y la manga del patrón de una camiseta, trazados en tinta, con la medida del dobladillo: ${dobladilloMm} milímetros">`,
    `<rect class="marcador" x="${MARGEN}" y="4" width="${f(archivo.length * 6.7)}" height="14"/>`,
    `<text class="rotulo" x="${MARGEN}" y="15">${archivo}</text>`,
    `<path class="pieza" data-pieza="${delantero.name}" d="${d(delantero.outline, tD)}"/>`,
    `<path class="pieza" data-pieza="${manga.name}" d="${d(manga.outline, tM)}"/>`,
    lineas(delantero, tD, 'grainline'), lineas(manga, tM, 'grainline'),
    lineas(delantero, tD, 'notch'), lineas(manga, tM, 'notch'),
    `<text class="rotulo" x="${f(MARGEN + 24)}" y="${f(oy + 0.2 * (cd.y1 - cd.y0) * s)}">DELANTERO · ×1</text>`,
    `<text class="rotulo" x="${f(mx + 24)}" y="${f(oy + 0.5 * (cm.y1 - cm.y0) * s)}">MANGA · ×2</text>`,
    `<g class="cota"><path d="M${f(hx0)} ${f(yc)} H${f(hx1)} M${f(hx0)} ${f(yc - 6)} v12 M${f(hx1)} ${f(yc - 6)} v12"/>`,
    `<rect x="${f((hx0 + hx1) / 2 - 70)}" y="${f(yc + 6)}" width="140" height="22" rx="3"/>`,
    `<text x="${f((hx0 + hx1) / 2)}" y="${f(yc + 21)}" text-anchor="middle">${dobladilloMm} mm · dobladillo</text></g>`,
    `</svg>`,
  ].join('\n') + '\n';
  return { svg, dobladilloMm };
}

if (process.argv[1] && fileURLToPath(import.meta.url) === process.argv[1]) {
  const dxf = readFileSync('tests/fixtures/garments/maria_patrones_samuel_2M.dxf', 'utf8');
  const { svg, dobladilloMm } = generarPatronSvg(dxf);
  writeFileSync('home/img/patron.svg', svg);
  console.log(`home/img/patron.svg · dobladillo ${dobladilloMm} mm`);
}
```

```ts
/** La home incrusta ficheros en su HTML (`<!-- incluir: ruta -->`) para que el
 *  SVG del hero herede los tokens y la fuente de la página: un <img> no puede. */
function incluirEnHome(): import('vite').Plugin {
  return {
    name: 'incluir-en-home',
    transformIndexHtml(html, ctx) {
      if (!ctx.filename.endsWith(`${path.sep}index.html`) || ctx.filename.includes(`${path.sep}app${path.sep}`)) return html;
      return html.replace(/<!-- incluir: ([\w/.-]+) -->/g, (_, rel) => fs.readFileSync(path.resolve(rel), 'utf8'));
    },
  };
}
```

```js
/** PNG → WebP con el codificador de Chromium, con recorte opcional (en px del PNG). */
async function aWebp(page, png, destino, { recorte, ancho, calidad = 0.82 } = {}) {
  const url = await page.evaluate(async ({ b64, recorte, ancho, calidad }) => {
    const img = await createImageBitmap(await (await fetch(`data:image/png;base64,${b64}`)).blob());
    const r = recorte ?? { x: 0, y: 0, w: img.width, h: img.height };
    const w = ancho ?? r.w, h = Math.round(r.h * (w / r.w));
    const c = new OffscreenCanvas(w, h); c.getContext('2d').drawImage(img, r.x, r.y, r.w, r.h, 0, 0, w, h);
    const blob = await c.convertToBlob({ type: 'image/webp', quality: calidad });
    return `data:image/webp;base64,${btoa(String.fromCharCode(...new Uint8Array(await blob.arrayBuffer())))}`;
  }, { b64: png.toString('base64'), recorte, ancho, calidad });
  await writeFile(path.join(REPO, destino), Buffer.from(url.split(',')[1], 'base64'));
  console.log('✓', destino);
}
```

```js
function pngAIco(png) { // ICO con una entrada PNG (válido desde Vista; todos los navegadores lo leen)
  const h = Buffer.alloc(22);
  h.writeUInt16LE(0, 0); h.writeUInt16LE(1, 2); h.writeUInt16LE(1, 4);       // reservado, tipo=icono, 1 imagen
  h.writeUInt8(32, 6); h.writeUInt8(32, 7); h.writeUInt8(0, 8); h.writeUInt8(0, 9);
  h.writeUInt16LE(1, 10); h.writeUInt16LE(32, 12);                           // planos, bpp
  h.writeUInt32LE(png.length, 14); h.writeUInt32LE(22, 18);                  // tamaño, offset
  return Buffer.concat([h, png]);
```
