// src/utils/cloth/editClassifier.ts
import type { PatternPiece } from '../../types/types';

/**
 * Type of pattern edit detected by comparing two piece arrays.
 *
 * - 'dimension': Only coordinate values changed (same topology). Safe for warm-start.
 * - 'topology': Piece count, point count, or segment types changed. Requires full restart.
 */
export type EditType = 'dimension' | 'topology';

/**
 * Classify a pattern edit by comparing previous and next piece arrays.
 *
 * A "dimension" edit keeps the same mesh topology (same piece count, same
 * number of outline segments per piece, same segment types). Only coordinates
 * changed — e.g. dragging a point, resizing an edge.
 *
 * A "topology" edit changes the mesh structure — adding/removing pieces,
 * adding/removing points, changing curve types. Requires full re-mesh.
 */
export function classifyEdit(prev: PatternPiece[], next: PatternPiece[]): EditType {
  if (prev.length !== next.length) return 'topology';

  for (let i = 0; i < prev.length; i++) {
    // Piece identity must match (no reordering)
    if (prev[i].id !== next[i].id) return 'topology';

    const prevOutline = prev[i].outline;
    const nextOutline = next[i].outline;

    // Same number of path segments
    if (prevOutline.length !== nextOutline.length) return 'topology';

    // Same segment types (M, L, C, Q, Z, etc.)
    for (let j = 0; j < prevOutline.length; j++) {
      if (prevOutline[j].type !== nextOutline[j].type) return 'topology';
      // Same number of values per segment (catches L→C type changes within same type count)
      if (prevOutline[j].values.length !== nextOutline[j].values.length) return 'topology';
    }
  }

  return 'dimension';
}
```

// src/utils/cloth/restLengthComputer.ts
/**
 * Compute Euclidean rest lengths for stretch constraint edges.
 *
 * Given vertex positions and edge pairs (stretchingIds), returns an array
 * of distances. Used to update solver constraints on dimension-only edits.
 */
export function computeRestLengths(
  vertices: Float32Array,
  stretchingIds: Int32Array,
): Float32Array {
  const numEdges = stretchingIds.length / 2;
  const lengths = new Float32Array(numEdges);
  for (let i = 0; i < numEdges; i++) {
    const id0 = stretchingIds[2 * i];
    const id1 = stretchingIds[2 * i + 1];
    const dx = vertices[id1 * 3] - vertices[id0 * 3];
    const dy = vertices[id1 * 3 + 1] - vertices[id0 * 3 + 1];
    const dz = vertices[id1 * 3 + 2] - vertices[id0 * 3 + 2];
    lengths[i] = Math.sqrt(dx * dx + dy * dy + dz * dz);
  }
  return lengths;
}
```

  useEffect(() => {
    const prevPieces = prevPiecesRef.current;
    prevPiecesRef.current = pieces;

    // Skip on first render or when no solver is active
    if (prevPieces.length === 0) return;
    if (simulationState !== 'running' && simulationState !== 'paused') return;

    const editType = classifyEdit(prevPieces, pieces);
    if (editType !== 'dimension') return; // topology change → let the existing useEffect handle it

    // Dimension edit detected — warm-start path
    // Cancel any pending warm-start timer
    if (warmStartTimerRef.current) {
      clearTimeout(warmStartTimerRef.current);
    }

    dispatch(setIsWarmStarting(true));

    // Debounce: wait 500ms of no further edits before triggering warm-start
    warmStartTimerRef.current = setTimeout(() => {
      // Re-mesh to get new vertex positions (same topology, new coordinates)
      // Then compute new rest lengths and send to solver
      const client = new SimulationInitClient();
      client.meshAllPieces(pieces, pieceParticleDistances, globalParticleDistance)
        .then(meshed => {
          client.dispose();

          // Compute new rest lengths from the re-meshed vertices
          const stretchIds = stretchingIdsRef.current;
          if (!stretchIds || meshed.length === 0) return;

          // Merge all piece vertices (same order as solver)
          const totalVerts = meshed.reduce((sum, p) => sum + p.vertices.length, 0);
          const mergedVerts = new Float32Array(totalVerts);
          let offset = 0;
          for (const p of meshed) {
            mergedVerts.set(p.vertices, offset);
            offset += p.vertices.length;
          }

          const newRestLengths = computeRestLengths(mergedVerts, stretchIds);

          // Send to active solver
          const activeClient = clientRef.current || wasmClientRef.current;
          if (activeClient && 'updateConstraints' in activeClient) {
            (activeClient as { updateConstraints: (rl: Float32Array) => void }).updateConstraints(newRestLengths);
          }

          // Also update pieceMeshes so static rendering matches
          setPieceMeshes(meshed.map(m => {
            const piece = pieces.find(p => p.id === m.pieceId);
            return {
              ...m,
              outline: (piece?.outline ?? []) as Array<{ type: string; values: number[] }>,
            };
          }));

          dispatch(setIsWarmStarting(false));
        })
        .catch(err => {
          client.dispose();
          console.warn('[GarmentSimulation] Warm-start re-mesh failed:', err);
          dispatch(setIsWarmStarting(false));
        });
    }, 500);

    return () => {
      if (warmStartTimerRef.current) {
        clearTimeout(warmStartTimerRef.current);
        warmStartTimerRef.current = null;
      }
    };

import { useEffect, useRef } from 'react';
import type { PatternPiece } from '../../types/types';
import { classifyEdit } from '../../utils/cloth/editClassifier';

interface GeometricPreviewOptions {
  pieces: PatternPiece[];
  simulationState: string;
  positionsRef: React.MutableRefObject<Float32Array | null>;
  onPreviewFrame: ((positions: Float32Array) => void) | null;
}

/**
 * Applies instant geometric vertex displacement when pattern dimensions change.
 *
 * This provides immediate visual feedback before the debounced warm-start
 * re-simulation kicks in. The displacement is approximate (proportional
 * scaling) but gives the user instant feedback.
 *
 * For v1, we simply scale vertex positions proportionally based on the
 * bounding box change of each piece's outline. This works well for
 * uniform scaling and point drags.
 */
export function useGeometricPreview({
  pieces,
  simulationState,
  positionsRef,
  onPreviewFrame,
}: GeometricPreviewOptions): void {
  const prevPiecesRef = useRef<PatternPiece[]>([]);

  useEffect(() => {
    const prev = prevPiecesRef.current;
    prevPiecesRef.current = pieces;

    if (prev.length === 0) return;
    if (simulationState !== 'running' && simulationState !== 'paused') return;
    if (!positionsRef.current || !onPreviewFrame) return;

    const editType = classifyEdit(prev, pieces);
    if (editType !== 'dimension') return;

    // Compute bounding box scale factors from outline changes
    const positions = positionsRef.current;
    const preview = new Float32Array(positions);

    // For each piece, compute 2D bounding box ratio
    let vertexOffset = 0;
    for (let pi = 0; pi < pieces.length; pi++) {
      const prevBB = outlineBounds(prev[pi].outline);
      const nextBB = outlineBounds(pieces[pi].outline);

      if (!prevBB || !nextBB) continue;

      const scaleX = prevBB.width > 0 ? nextBB.width / prevBB.width : 1;
      const scaleZ = prevBB.height > 0 ? nextBB.height / prevBB.height : 1;

      // Count vertices for this piece (we don't have per-piece counts here,
      // so apply uniform scale to all vertices — imprecise but fast)
      // In practice, the warm-start will correct this within 1-2s
      const numVerts = positions.length / 3 / pieces.length;

      for (let v = 0; v < numVerts; v++) {
        const idx = (vertexOffset + v) * 3;
        // Scale relative to piece center
        const cx = prevBB.cx;
        const cz = prevBB.cy; // 2D y maps to 3D z

        preview[idx] = cx + (positions[idx] - cx) * scaleX;
        // y (vertical) stays unchanged — gravity handles it
        preview[idx + 2] = cz + (positions[idx + 2] - cz) * scaleZ;
      }
      vertexOffset += numVerts;
    }

    onPreviewFrame(preview);
  }, [pieces, simulationState, positionsRef, onPreviewFrame]);
}

interface BoundingBox {
  cx: number; cy: number;
  width: number; height: number;
}

function outlineBounds(outline: Array<{ type: string; values: number[] }>): BoundingBox | null {
  let minX = Infinity, minY = Infinity, maxX = -Infinity, maxY = -Infinity;
  let hasPoints = false;

  for (const cmd of outline) {
    const vals = cmd.values;
    // Extract x,y pairs from values
    for (let i = 0; i < vals.length - 1; i += 2) {
      const x = vals[i], y = vals[i + 1];
      minX = Math.min(minX, x);
      maxX = Math.max(maxX, x);
      minY = Math.min(minY, y);
      maxY = Math.max(maxY, y);
      hasPoints = true;
    }
  }

  if (!hasPoints) return null;
  return {
    cx: (minX + maxX) / 2,
    cy: (minY + maxY) / 2,
    width: maxX - minX,
    height: maxY - minY,
  };
}
