> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Export the full 3D scene (garment + avatar + PBR materials) as glTF/GLB and USDZ with user-configurable options.

**Architecture:** A scene collector traverses the live Three.js scene, clones garment and avatar meshes into a clean group (skipping lights, gizmos, ground, grid), applies a unit-scale factor, then hands the group to the appropriate exporter. The UI lives in the existing RenderSettingsPanel export section.

**Tech Stack:** Three.js GLTFExporter, Three.js USDZExporter, React, Redux

```typescript
/**
 * Scene Collector — builds a clean THREE.Group for 3D export.
 *
 * Traverses the live Three.js scene and clones only garment + avatar meshes,
 * skipping lights, helpers, gizmos, ground, grid, contact shadows.
 *
 * @module sceneCollector
 */

import * as THREE from 'three';

export interface CollectOptions {
  /** The garment mesh ref (window.__OPENCLO_CLOTH_MESH__). If null, garment is excluded. */
  garmentMesh: THREE.Mesh | null;
  /** Include the avatar model (default true) */
  includeAvatar?: boolean;
  /** Unit conversion multiplier applied to the group (default 1.0 = mm) */
  scale?: number;
}

/**
 * Collect exportable objects from the live scene into a clean group.
 */
export function collectExportScene(
  scene: THREE.Scene,
  options: CollectOptions,
): THREE.Group {
  const { garmentMesh, includeAvatar = true, scale = 1.0 } = options;
  const group = new THREE.Group();

  // 1. Add garment mesh clone
  if (garmentMesh) {
    group.add(garmentMesh.clone(true));
  }

  // 2. Add avatar — find groups containing SkinnedMesh
  if (includeAvatar) {
    for (const child of scene.children) {
      if (child === garmentMesh) continue;
      let hasSkinnedMesh = false;
      child.traverse(obj => {
        if ((obj as THREE.SkinnedMesh).isSkinnedMesh) hasSkinnedMesh = true;
      });
      if (hasSkinnedMesh) {
        group.add(child.clone(true));
      }
    }
  }

  // 3. Apply scale
  if (scale !== 1.0) {
    group.scale.set(scale, scale, scale);
    group.updateMatrixWorld(true);
  }

  return group;
}
```

```typescript
/**
 * USDZ Exporter — export 3D scene as Apple USDZ for AR Quick Look.
 *
 * Uses three.js built-in USDZExporter. Maps PBR materials (MeshPhysicalMaterial)
 * to USD Preview Surface automatically.
 *
 * @module usdzExporter
 */

import * as THREE from 'three';
import { USDZExporter } from 'three/examples/jsm/exporters/USDZExporter.js';

export interface USDZExportOptions {
  /** Filename without extension (default 'openclo-export') */
  filename?: string;
}

/**
 * Export a THREE.js object as USDZ.
 * Returns the USDZ as a Uint8Array.
 */
export async function exportToUSDZ(
  object: THREE.Object3D,
  options: USDZExportOptions = {},
): Promise<Uint8Array> {
  const exporter = new USDZExporter();
  return await exporter.parse(object);
}

/**
 * Download a USDZ file.
 */
export async function downloadUSDZ(
  object: THREE.Object3D,
  options: USDZExportOptions = {},
): Promise<void> {
  const { filename = 'openclo-export' } = options;
  const data = await exportToUSDZ(object, options);
  const blob = new Blob([data], { type: 'model/vnd.usdz+zip' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = `${filename}.usdz`;
  a.click();
  URL.revokeObjectURL(url);
}
```

```typescript
const handleExport3D = useCallback(async (
  format: '3d-glb' | '3d-gltf' | '3d-usdz',
  includeAvatar: boolean,
  scale: number,
) => {
  const container = containerRef.current;
  if (!container) return;
  const refs = container as unknown as Record<string, unknown>;
  const sc = refs.__scene as THREE.Scene | undefined;
  if (!sc) return;

  // Get garment mesh from window global (set by GarmentSimulation)
  // eslint-disable-next-line @typescript-eslint/no-explicit-any
  const garmentMesh = (window as any).__OPENCLO_CLOTH_MESH__ as THREE.Mesh | undefined;

  const group = collectExportScene(sc, {
    garmentMesh: garmentMesh || null,
    includeAvatar,
    scale,
  });

  const timestamp = Date.now();
  if (format === '3d-usdz') {
    await downloadUSDZ(group, { filename: `openclo-${timestamp}` });
  } else {
    const binary = format === '3d-glb';
    await downloadGLTF(group, {
      binary,
      scale: 1, // scale already applied by collector
      filename: `openclo-${timestamp}`,
    });
  }
}, []);
```
